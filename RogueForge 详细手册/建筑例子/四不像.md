using System.Collections.Generic;
using RogueForge;
using RogueLibsCore;
using System;
using UnityEngine;

/// <summary>
/// 示例建筑：支持交互（翻找/识别/购买）、购买列表、容器、破碎、生成。
/// </summary>
[ObjectName("ExampleBuilding")]
public class ExampleBuilding : CustomObjectReal, IObjectInteraction, IBuyable, IBuildingSpawner, IBreakOpen, IObjectContainer
{
    [RLSetup]
    public static void Setup()
    {
        CustomObjects.CreateCustomObject<ExampleBuilding>()
            .WithName(new CustomNameInfo { Chinese = "示例建筑" })
            .WithDescription(new CustomNameInfo { Chinese = "这是一个示例建筑。" })
            .WithSprites(
				Mod.Properties.Resources.ExampleBuildingSprite_N,   // 北
				Mod.Properties.Resources.ExampleBuildingSprite_E,   // 东
				Mod.Properties.Resources.ExampleBuildingSprite_S,   // 南
				Mod.Properties.Resources.ExampleBuildingSprite_W)   // 西
            .WithScale(1f)
            .WithCloneSource("Chair");
    }

    public override void SetVars()
    {
        base.damageAccumulates = true;
        base.damageThreshold = 30;
        base.hackable = true;
        this.bulletsCanPass = true;
        this.noInterest = true;
        this.strikeOnHit = true;
        this.noShadow = true;
        this.collidersDontDisappear = true;
        this.damageImmediateOnClient = true;
        this.importantToClient = true;
        base.fireProof = true;
        base.interactable = true;
        this.pickUppable = false;
    }

    // ===== IBuyable =====
    public List<InvItem> GetBuyItems()
    {
        var items = new List<InvItem>();

        var banana = new InvItem();
        banana.invItemName = "Banana";
        banana.invItemCount = 1;
        banana.ItemSetup(notNew: true);
        banana.itemValue = 48484;
        items.Add(banana);

        var peel = new InvItem();
        peel.invItemName = "BananaPeel";
        peel.invItemCount = 2;
        peel.ItemSetup(notNew: true);
        peel.itemValue = 0;
        items.Add(peel);

        return items;
    }

    public void OnItemBought(InvItem item, Agent buyer)
    {
        CustomBuildingsPlugin.Logger.LogInfo($"[ExampleBuilding] 玩家选中商品: {item?.invItemName}, 购买者={buyer?.agentName}");
        bool success = this.PurchaseItem(this, buyer!, item!);
        CustomBuildingsPlugin.Logger.LogInfo($"[ExampleBuilding] 购买结果: {(success ? "成功" : "失败")}");
    }

    // 自定义价格显示和槽位颜色
    public override string? PriceOverride1 => "香蕉";
    public override string? PriceOverride2 => "香蕉皮";
    public override Color? PriceOverrideColor1 => new Color32(0x49, 0x89, 0x49, 0xC0);

    // ===== IObjectInteraction =====
    public void SetupInteractions(SimpleInteractionProvider h)
    {
        // 远程黑客交互
        if (h.Helper.interactingFar)
        {
            h.AddButton("RogueForge_启动自毁", m =>
            {
                if (m.Object is ObjectReal target)
                    target.HackExplode(m.Agent);
            });
            h.AddButton("RogueForge_吐出货物", m =>
            {
                this.SpillContainerItems(m.Agent);
            });
            return;
        }

        // 近距离交互
        h.AddButton("RogueForge_翻找", m =>
        {
            StartDelayedAction(m.Agent, 2f, "RogueForge_翻找", () =>
            {
                this.OpenContainer((ObjectReal)m.Object, m.Agent);
                CustomBuildingsPlugin.Logger.LogInfo("[ExampleBuilding] 翻找完成，打开容器");
            });
        });

        h.AddButton("RogueForge_识别物品", m =>
        {
            m.Object.ShowUseOn("RogueForge_SayItem");
            CustomBuildingsPlugin.Logger.LogInfo("[ExampleBuilding] 识别物品按钮点击");
        });

        h.AddButton("RogueForge_购买", m =>
        {
            this.OpenBuyChest((ObjectReal)m.Object, m.Agent);
        });
    }

    public override bool UseItemOnObject(InvItem item, int slotNum, string combineType, string useOnType)
    {
        if (useOnType != "RogueForge_SayItem") return false;
        if (combineType != "Combine") return true; // UI 高亮检测

        if (interactingAgent != null)
            interactingAgent.Say(item.invItemName);
        return true;
    }

    // ===== 容器 =====
    protected override void FillContainer(InvDatabase database)
    {
        database.AddItem("BananaPeel", 500);
    }

    public override void OnContainerOpened()
    {
        base.OnContainerOpened();
        CustomBuildingsPlugin.Logger.LogInfo("[ExampleBuilding] 容器已打开");
    }

    // ===== IBreakOpen =====
    public void OnBreakOpen(PlayfieldObject damagerObject)
    {
        CustomBuildingsPlugin.Logger.LogInfo("[ExampleBuilding] 破碎回调，物品将洒落");
    }

    // ===== 黑客完成 =====
    protected override void OnHackingComplete(Agent hacker)
    {
        base.OnHackingComplete(hacker);
        this.ShowObjectButtons();
    }

    // ===== 生成 =====
    public void OnLevelSpawn(LoadLevel level)
    {
        Chunk? exit = KMap.GetEndingChunk();
        if (exit == null) return;
        Vector2 pos = KMap.FindEmptySpotByChunkID(exit.chunkID, maxRetries: 500);
        if (pos == Vector2.zero) return;
        KMap.SpawnObject("ExampleBuilding", pos);
    }
}
